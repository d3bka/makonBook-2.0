# Hollihop smart-sync settings
# Add/update these in your CURRENT .env. Keep your existing API URL/key/secret.

SCHOOL_DATA_PROVIDER=hollihop
HOLLIHOP_ENABLED=true
HOLLIHOP_MODE=users

# Import only ordinary GROUP learning units.
HOLLIHOP_ALLOWED_LEARNING_TYPES=GROUP

# Lightweight delta cycle. It is inert until the explicit initial sync succeeds.
HOLLIHOP_SYNC_INTERVAL_MINUTES=5

# Teachers are a small endpoint in your tenant; existing linked teachers are checked every 5 min.
HOLLIHOP_TEACHER_POLL_MINUTES=5
HOLLIHOP_MANAGER_POLL_MINUTES=15

# GetEdUnitStudents is heavy. A full relation-only sweep intentionally runs only
# several times/day and requests NO Days/payments. Use the Manager button for an
# immediate membership reconciliation when needed.
HOLLIHOP_MEMBERSHIP_SWEEP_MINUTES=240

# Safety catalog/profile sweeps for changes that Hollihop does not expose via
# a reliable lastUpdated/webhook signal.
HOLLIHOP_STUDENT_PROFILE_SWEEP_MINUTES=360
HOLLIHOP_EDUNIT_CATALOG_SWEEP_MINUTES=360

# Small overlap protects against timestamp/race boundaries.
HOLLIHOP_CHECKPOINT_OVERLAP_SECONDS=120

# Hollihop limit reported by support: 600 requests / 30 seconds.
# 0.10 sec is deliberately conservative for bursty targeted dependency fetches.
HOLLIHOP_MIN_REQUEST_INTERVAL=0.10

# IMPORTANT while Eskiz is in test mode:
# automatic reconciliation/webhooks MUST NOT send credentials/SMS.
HOLLIHOP_AUTO_SEND_CREDENTIALS=false
HOLLIHOP_SEND_CREDENTIALS=false
